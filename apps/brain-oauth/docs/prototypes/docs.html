const base64url = (bytes) =>
  btoa(String.fromCharCode(...new Uint8Array(bytes)))
    .replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');

const verifier = base64url(crypto.getRandomValues(new Uint8Array(32)));
const challenge = base64url(
  await crypto.subtle.digest('SHA-256', new TextEncoder().encode(verifier))
);

设计说明

文档是阅读型页面：背景用「安静版」，正文直接放在底色上，只有表格和代码块用卡片承载，保证长文好读。标题细体，正文 14px、行高 1.9。

构成

  • 页头：logo + 竖线 +「集成文档」，导航「测试台 / 控制台」；访客显示黑色「登录」胶囊，已登录显示头像。
  • 快捷入口：三张卡片——站内测试完整流程、OpenAPI 参考、管理 OAuth 应用。
  • 左侧目录：竖线 + 文字，当前章节竖线变成品牌渐变；真实页面滚动时自动高亮并吸顶。手机端隐藏。
  • 章节：概览（机密 / 公共客户端两张小卡）→ 授权码流程（编号时间线）→ 端点一览（表格）→ 授权请求（参数表 + 示例）→ 令牌交换（授权码 / 刷新令牌标签切换 + 响应示例）→ PKCE（JS 示例）→ 用户信息 → 错误响应（错误码表）。
  • 代码块：右上角显示语言和复制按钮，复制后轻提示。
  • 表头改为中文：原来端点表的表头是英文硬编码。
